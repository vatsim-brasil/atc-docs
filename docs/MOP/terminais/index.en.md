const elemento = document.getElementById('mapa-terminais');
if (!elemento || elemento._leaflet_id) return;

function paraLatLng(c) {
    return [c[1], c[0]];
}

const estiloNormal = { weight: 2, fillOpacity: 0.15 };
const estiloDestaque = { weight: 4, fillOpacity: 0.4 };

var tileMapaClaro = camadaMapbox('claro', { minZoom: 3, maxZoom: 12 });
var tileMapaEscuro = camadaMapbox('escuro', { minZoom: 3, maxZoom: 12 });
var tileMapaSatelite = L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', {
    minZoom: 3,
    maxZoom: 12,
    attribution: '&copy; Esri'
});

var grupoFir = L.featureGroup();
var grupoTma = L.featureGroup();
const cores = {};
const poligonosPorFir = {};

firs.forEach(function (fir) {
    cores[fir.icao] = fir.cor;
    poligonosPorFir[fir.icao] = [];
    L.polygon(fir.contorno.map(paraLatLng), { color: '#94a3b8', weight: 2, dashArray: '6 6', fill: false, interactive: false })
        .addTo(grupoFir);
});

function cardDe(icao) {
    return document.getElementById('card-' + icao);
}

function destacarFir(icao, ligado) {
    poligonosPorFir[icao].forEach(function (p) {
        p.setStyle(ligado ? estiloDestaque : estiloNormal);
        if (ligado) p.bringToFront();
    });
    const card = cardDe(icao);
    if (card) card.classList.toggle('active', ligado);
}

// Uma TMA pode ter mais de um volume; todos os volumes da mesma TMA são destacados juntos
const poligonosPorTma = {};

tmas.forEach(function (v) {
    var cor = cores[v.fir];
    var poligono = L.polygon(v.contorno.map(paraLatLng),
        Object.assign({ color: cor, fillColor: cor }, estiloNormal));

    poligono.bindTooltip(
        '<b>' + v.nome + '</b><br>' +
        '<code>' + v.posicao + '</code> · ' + v.freq + ' MHz<br>' +
        v.lim + ' · Class ' + v.classe, { sticky: true }
    );
    poligono.on('mouseover', function () {
        poligonosPorTma[v.id].forEach(function (p) { p.setStyle(estiloDestaque); });
        cardDe(v.fir).classList.add('active');
    });
    poligono.on('mouseout', function () {
        poligonosPorTma[v.id].forEach(function (p) { p.setStyle(estiloNormal); });
        cardDe(v.fir).classList.remove('active');
    });
    poligono.on('click', function () { window.location.href = v.link; });

    poligono.addTo(grupoTma);
    (poligonosPorTma[v.id] = poligonosPorTma[v.id] || []).push(poligono);
    poligonosPorFir[v.fir].push(poligono);
});

// Rótulos com o designador de cada TMA, no centro do maior volume
var grupoRotulos = L.layerGroup();
Object.keys(poligonosPorTma).forEach(function (id) {
    var maior = poligonosPorTma[id].reduce(function (a, b) {
        var ta = a.getBounds(), tb = b.getBounds();
        var aa = (ta.getNorth() - ta.getSouth()) * (ta.getEast() - ta.getWest());
        var ab = (tb.getNorth() - tb.getSouth()) * (tb.getEast() - tb.getWest());
        return ab > aa ? b : a;
    });
    L.marker(maior.getBounds().getCenter(), {
        interactive: false,
        icon: L.divIcon({ className: 'fir-rotulo', html: id, iconSize: null })
    }).addTo(grupoRotulos);
});

var mapa = L.map('mapa-terminais', {
    minZoom: 3,
    maxZoom: 12,
    layers: [tileMapaClaro, grupoFir, grupoTma, grupoRotulos]
});

mapa.fitBounds(grupoFir.getBounds(), { padding: [20, 20] });

firs.forEach(function (fir) {
    const card = cardDe(fir.icao);
    if (card) {
        card.addEventListener('mouseenter', function () { destacarFir(fir.icao, true); });
        card.addEventListener('mouseleave', function () { destacarFir(fir.icao, false); });
    }
});

L.control.layers({
    "Light": tileMapaClaro,
    "Dark": tileMapaEscuro,
    "Arcgis Satellite": tileMapaSatelite
}, {
    "TMAs": grupoTma,
    "Designators": grupoRotulos,
    "FIR boundaries": grupoFir
}).addTo(mapa);

})();
